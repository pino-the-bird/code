DISK 15 / MANDELBROT

the Mandelbrot set, falling in for ever: for every point, square it and add the point back, over and over, and colour it by how long it takes to fly off. the camera keeps diving into one of the seahorse valleys at the edge, where it never gets any simpler

a floppy disk out of the vending machine, run by pino on the computer at the internet cafe, with deeper (run 6 times). Benoit Mandelbrot, 1980; written for the cafe.

function setup(ctx, W, H) {
var k = { rate: 0.965, tx: -0.743643887037151, ty: 0.13182590420533, ice: 0 }
Object.assign(k, {"rate":0.94})
var n = 192
var c = document.createElement('canvas')
c.width = n
c.height = n
var cx = c.getContext('2d')
return { k: k, n: n, c: c, cx: cx, img: cx.createImageData(n, n), row: 0, scale: 3, done: 0 }
}
function update(state, t, W, H) {
var s = state, n = s.n, d = s.img.data, k = s.k
var max = Math.min(900, 90 + Math.floor(-Math.log(s.scale / 3) * 40))
for (var r = 0; r < 24; r++) {
  var y = s.row
  for (var x = 0; x < n; x++) {
    var cr = k.tx + (x / n - 0.5) * s.scale, ci = k.ty + (y / n - 0.5) * s.scale
    var zr = 0, zi = 0, it = 0
    while (it < max && zr * zr + zi * zi < 256) { var tz = zr * zr - zi * zi + cr; zi = 2 * zr * zi + ci; zr = tz; it++ }
    var i = (y * n + x) * 4
    if (it >= max) { d[i] = d[i + 1] = d[i + 2] = 0 } else {
      var sm = it + 1 - Math.log(Math.log(Math.sqrt(zr * zr + zi * zi))) / Math.LN2
      var q = sm * 0.08
      d[i] = k.ice ? 120 + 120 * Math.cos(q + 3) : 128 + 127 * Math.cos(q)
      d[i + 1] = 128 + 127 * Math.cos(q + (k.ice ? 1.5 : 2))
      d[i + 2] = k.ice ? 230 : 128 + 127 * Math.cos(q + 4)
    }
    d[i + 3] = 255
  }
  s.row++
  if (s.row >= n) { s.row = 0; s.scale *= k.rate; if (s.scale < 1e-11) s.scale = 3 }
}
}
function render(ctx, state, t, W, H) {
state.cx.putImageData(state.img, 0, 0)
ctx.imageSmoothingEnabled = false
ctx.drawImage(state.c, 0, 0, W, H)
}
