DISK 02 / RAYCAST

a walk round your own town in first person, the way the 1992 shooters drew corridors: no 3D at all, one ray thrown out per column of the screen until it hits a wall, and the wall drawn taller the nearer it is. the streets are a map of the town, the gold wall is the shrine

a floppy disk out of the vending machine, run by pino on the computer at the internet cafe. the raycasting of Wolfenstein 3D (id Software, 1992), as Lode Vandevenne taught it; written again for the cafe, on a map of the town.

function setup(ctx, W, H) {
var k = { fov: 0.66, speed: 0.035, night: 0 }
/*TWEAK*/
var map = ['1111111111111111', '1000000000000001', '1022200300222001', '1020000000002001', '1020044400002001', '1000040000000001', '1030040005000301', '1000000000000001', '1022200111022201', '1020000101000201', '1020000000000201', '1000033000330001', '1044000000004401', '1040002220000401', '1000000000000001', '1111111111111111']
// a loop of the streets: along the middle, up the east side, back along the top, down the west, along the bottom
var path = [[1.5, 7.5], [14.5, 7.5], [14.5, 1.5], [1.5, 1.5], [1.5, 14.5], [14.5, 14.5], [14.5, 7.5]]
return { k: k, map: map, path: path, seg: 0, u: 0, x: 1.5, y: 7.5, a: 0 }
}
function update(state, t, W, H) {
var p = state.path, s = state.seg, k = state.k
var ax = p[s][0], ay = p[s][1], bx = p[(s + 1) % p.length][0], by = p[(s + 1) % p.length][1]
var len = Math.hypot(bx - ax, by - ay)
state.u += k.speed / len
if (state.u >= 1) { state.u = 0; state.seg = (s + 1) % p.length; return }
state.x = ax + (bx - ax) * state.u
state.y = ay + (by - ay) * state.u
var want = Math.atan2(by - ay, bx - ax) + Math.sin(t / 90) * 0.35
var d = want - state.a
while (d > Math.PI) d -= Math.PI * 2
while (d < -Math.PI) d += Math.PI * 2
state.a += d * 0.06
}
function render(ctx, state, t, W, H) {
var k = state.k, map = state.map
var sky = ctx.createLinearGradient(0, 0, 0, H / 2)
sky.addColorStop(0, k.night ? '#05060f' : '#2b1d3a')
sky.addColorStop(1, k.night ? '#1b2340' : '#e98a5c')
ctx.fillStyle = sky
ctx.fillRect(0, 0, W, H / 2)
var ground = ctx.createLinearGradient(0, H / 2, 0, H)
ground.addColorStop(0, '#1a1716')
ground.addColorStop(1, '#4a4440')
ctx.fillStyle = ground
ctx.fillRect(0, H / 2, W, H / 2)
var dx = Math.cos(state.a), dy = Math.sin(state.a)
var px = -dy * k.fov, py = dx * k.fov
var cols = 256, cw = W / cols
var base = { 1: [150, 70, 50], 2: [130, 130, 135], 3: [60, 150, 150], 4: [50, 110, 55], 5: [230, 180, 60] }
for (var c = 0; c < cols; c++) {
  var cam = 2 * c / cols - 1
  var rx = dx + px * cam, ry = dy + py * cam
  var mx = Math.floor(state.x), my = Math.floor(state.y)
  var ddx = Math.abs(1 / rx), ddy = Math.abs(1 / ry)
  var sx = rx < 0 ? -1 : 1, sy = ry < 0 ? -1 : 1
  var sdx = (rx < 0 ? state.x - mx : mx + 1 - state.x) * ddx
  var sdy = (ry < 0 ? state.y - my : my + 1 - state.y) * ddy
  var side = 0, cell = '0', guard = 0
  while (cell === '0' && guard++ < 64) {
    if (sdx < sdy) { sdx += ddx; mx += sx; side = 0 } else { sdy += ddy; my += sy; side = 1 }
    cell = (map[my] || '1')[mx] || '1'
  }
  var dist = side === 0 ? sdx - ddx : sdy - ddy
  var hgt = Math.min(H * 4, H / Math.max(0.05, dist))
  var hit = side === 0 ? state.y + dist * ry : state.x + dist * rx
  hit -= Math.floor(hit)
  var col = base[cell] || base[1]
  var shade = (side ? 0.72 : 1) * Math.max(0.15, 1 - dist / 14)
  // windows on the shop glass, courses of brick, and the shrine glowing
  if (cell === '3' && (hit * 4) % 1 < 0.7) shade *= 1.35
  if (cell === '1' && Math.floor(hit * 8) % 2) shade *= 0.9
  if (cell === '5') shade *= 1 + 0.25 * Math.sin(t / 10)
  if (k.night && cell === '3' && (hit * 4) % 1 < 0.7) col = [240, 200, 120]
  ctx.fillStyle = 'rgb(' + Math.min(255, col[0] * shade | 0) + ',' + Math.min(255, col[1] * shade | 0) + ',' + Math.min(255, col[2] * shade | 0) + ')'
  ctx.fillRect(c * cw, (H - hgt) / 2, cw + 0.5, hgt)
}
