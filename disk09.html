DISK 09 / MICROGRAD

a tiny neural network that learns, from nothing, a thing no single line can: XOR, true when exactly one of two inputs is on. every number keeps a note of how it was made, so the program can walk the notes backwards and work out how to nudge each weight. it prints the network getting less wrong

a floppy disk out of the vending machine, run by pino on the computer at the internet cafe. after Andrej Karpathy's micrograd (MIT licence); written again in JavaScript for the cafe.

var seed = 42
function rand() { seed = (seed * 1664525 + 1013904223) % 4294967296; return seed / 4294967296 * 2 - 1 }
function V(data, kids) { this.data = data; this.grad = 0; this.kids = kids || []; this.back = null }
function val(x) { return x instanceof V ? x : new V(x) }
V.prototype.add = function (o) { o = val(o); var a = this, out = new V(a.data + o.data, [a, o]); out.back = function () { a.grad += out.grad; o.grad += out.grad }; return out }
V.prototype.mul = function (o) { o = val(o); var a = this, out = new V(a.data * o.data, [a, o]); out.back = function () { a.grad += o.data * out.grad; o.grad += a.data * out.grad }; return out }
V.prototype.tanh = function () { var a = this, y = Math.tanh(a.data), out = new V(y, [a]); out.back = function () { a.grad += (1 - y * y) * out.grad }; return out }
V.prototype.backward = function () {
  var order = [], seen = new Set()
  var visit = function (v) { if (seen.has(v)) return; seen.add(v); v.kids.forEach(visit); order.push(v) }
  visit(this)
  this.grad = 1
  for (var i = order.length - 1; i >= 0; i--) if (order[i].back) order[i].back()
}
function layer(nin, nout, last) {
  var ns = []
  for (var i = 0; i < nout; i++) { var w = []; for (var j = 0; j < nin; j++) w.push(new V(rand())); ns.push({ w: w, b: new V(0) }) }
  return { ns: ns, last: last }
}
var net = [layer(2, 4), layer(4, 4), layer(4, 1, true)]
function forward(x) {
  var h = x.map(val)
  for (var l = 0; l < net.length; l++) {
    h = net[l].ns.map(function (n) { var s = n.b; for (var j = 0; j < n.w.length; j++) s = s.add(n.w[j].mul(h[j])); return s.tanh() })
  }
  return h[0]
}
var params = []
net.forEach(function (l) { l.ns.forEach(function (n) { n.w.forEach(function (w) { params.push(w) }); params.push(n.b) }) })
var xs = [[0, 0], [0, 1], [1, 0], [1, 1]], ys = [-1, 1, 1, -1]
print('MICROGRAD  //  a network of ' + params.length + ' numbers learning XOR')
print('')
for (var step = 0; step <= 200; step++) {
  var loss = new V(0)
  for (var i = 0; i < 4; i++) { var d = forward(xs[i]).add(-ys[i]); loss = loss.add(d.mul(d)) }
  params.forEach(function (p) { p.grad = 0 })
  loss.backward()
  params.forEach(function (p) { p.data -= 0.08 * p.grad })
  if (step % 25 === 0) {
    var bar = ''
    for (var b = 0; b < Math.min(40, Math.round(loss.data * 10)); b++) bar += '#'
    print('step ' + (step < 10 ? '  ' : step < 100 ? ' ' : '') + step + '  wrong by ' + loss.data.toFixed(4) + '  ' + bar)
  }
}
print('')
for (var q = 0; q < 4; q++) print(xs[q].join(' xor ') + '  ->  ' + forward(xs[q]).data.toFixed(3) + '   (wanted ' + ys[q] + ')')
