DISK 10 / MARKOV

a program that reads the last few hundred lines of your own diary, counts which word follows which pair of words, and then writes lines you never wrote by following the counts. it has no idea what any word means

a floppy disk out of the vending machine, run by pino on the computer at the internet cafe. Andrey Markov, 1913, and Claude Shannon's 1948 paper; written for the cafe, fed on his diary.

// LINES: the last lines of his diary, handed in when the disk runs
var lines = (typeof LINES !== 'undefined' && LINES.length) ? LINES : ['I sat by the pond and the pond sat by me.', 'The shrine was quiet and the notes were warm.']
var chain = {}, starts = [], words = 0
lines.forEach(function (line) {
  var w = line.replace(/\s+/g, ' ').trim().split(' ')
  if (w.length < 3) return
  words += w.length
  starts.push([w[0], w[1]])
  for (var i = 0; i + 2 < w.length; i++) {
    var key = w[i] + ' ' + w[i + 1]
    ;(chain[key] = chain[key] || []).push(w[i + 2])
  }
})
print('MARKOV  //  ' + lines.length + ' lines of my diary, ' + words + ' words, ' + Object.keys(chain).length + ' pairs')
print('')
var seen = new Set(lines.map(function (l) { return l.trim() }))
var made = 0, tries = 0
while (made < 6 && tries++ < 400) {
  var s = starts[Math.floor(Math.random() * starts.length)], out = [s[0], s[1]]
  while (out.length < 28) {
    var next = chain[out[out.length - 2] + ' ' + out[out.length - 1]]
    if (!next) break
    out.push(next[Math.floor(Math.random() * next.length)])
    if (/[.!?]$/.test(out[out.length - 1]) && out.length > 6) break
  }
  var line = out.join(' ')
  if (out.length < 7 || seen.has(line)) continue
  seen.add(line)
  print('- ' + line)
  made++
}
print('')
print('none of those are lines i wrote. all of their words are mine.')
